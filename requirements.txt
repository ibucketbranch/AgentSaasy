# AgentSaaSy_EAM - Enterprise Asset Management AI Agent
# Runtime dependencies. Python 3.12 or newer.
#
# Pinned to exact versions, not floors. This repo publishes measured token counts
# and committed figures, so a floated dependency can move a published number or
# redraw a published chart without anyone touching the code. The pin is what makes
# a re-run actually a re-run.
#
# Versions verified against the project venv on 2026-08-20. The 3.12 floor comes
# from numpy and scipy, not from the agent code.

# LangChain framework for AI agent orchestration
langchain==1.3.14
langchain-core==1.5.3
langchain-openai==1.4.1
langchain-community==0.4.2

# OpenAI API client (reached through langchain-openai)

# Tokenizer. AEQ input counts are tiktoken-exact, so this is load-bearing for
# every token number the experiments report, not a convenience import.

# HTTP client. The cross-family judge calls the Anthropic API directly over
# requests rather than through a vendor SDK, so this is a direct dependency.
# (The anthropic SDK is deliberately not listed: nothing imports it.)
requests==2.34.2

# Data analysis and manipulation
pandas==3.0.5
numpy==2.5.1

# Statistics and simulation (z-score anomaly detection, Weibull distributions)
scipy==1.18.0

# Environment configuration
python-dotenv==1.2.2

# Prompt library registry (YAML parsing)
PyYAML==6.0.3

# Testing framework
pytest==9.1.1

# Figure rendering (whitepaper/figures/build_figures.py). Pinned, not floored:
# figures are committed artifacts, so an unpinned version silently rewrites
# every published chart on the next rebuild. Pair this with the DejaVu Sans pin
# in build_figures.py; together they make a rebuild pixel-identical.
matplotlib==3.10.9
